theta = 0.0
baseline = 0.0
for step in range(2000):
    p = 1 / (1 + np.exp(-theta))
    a = rng.random() < p
    r = (11.0 if a else 10.0) + rng.normal()
    theta += 0.05 * (r - baseline) * (a - p)
    baseline += 0.01 * (r - baseline)
